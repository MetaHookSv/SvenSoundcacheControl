cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_policy(SET CMP0091 NEW)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    project(SvenSoundcacheControlTests LANGUAGES CXX)
    enable_testing()
endif()
add_executable(SvenSoundcacheControlTests soundcache_control_tests.cpp)
target_include_directories(SvenSoundcacheControlTests PRIVATE ../src)
target_compile_features(SvenSoundcacheControlTests PRIVATE cxx_std_11)
add_test(NAME SvenSoundcacheControl.Behavior COMMAND SvenSoundcacheControlTests)

if(TARGET SvenSoundcacheControl)
    add_executable(SvenSoundcacheControlHookTests hook_tests.cpp
        "${METAHOOK_SOURCE_PATH}/include/HLSDK/common/interface.cpp")
    get_target_property(_soundcache_includes SvenSoundcacheControl INCLUDE_DIRECTORIES)
    target_include_directories(SvenSoundcacheControlHookTests PRIVATE ${_soundcache_includes})
    target_compile_features(SvenSoundcacheControlHookTests PRIVATE cxx_std_11)
    add_test(NAME SvenSoundcacheControl.Hooks COMMAND SvenSoundcacheControlHookTests)
endif()
